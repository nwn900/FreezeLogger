cmake_minimum_required(VERSION 3.24)
project(FreezeLogger VERSION 0.3.6 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(FREELOGGER_SYMBOLS_MODE "required" CACHE STRING "Release symbol mode: required or none")
set_property(CACHE FREELOGGER_SYMBOLS_MODE PROPERTY STRINGS required none)
if(NOT FREELOGGER_SYMBOLS_MODE STREQUAL "required" AND NOT FREELOGGER_SYMBOLS_MODE STREQUAL "none")
  message(FATAL_ERROR "FREELOGGER_SYMBOLS_MODE must be required or none")
endif()
if(FREELOGGER_SYMBOLS_MODE STREQUAL "required" AND NOT MSVC)
  message(FATAL_ERROR "Required PDB packages need an MSVC-compatible compiler; use symbols mode none explicitly otherwise")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0603)
add_library(FreezeLogger SHARED src/plugin.cpp)
target_link_libraries(FreezeLogger PRIVATE shell32 ole32)
add_executable(FreezeLoggerWatch WIN32 src/watch.cpp src/capture.cpp src/frames.cpp src/history.cpp src/storage.cpp src/lifetime.cpp src/context.cpp)
target_link_libraries(FreezeLoggerWatch PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32 tdh)
add_executable(FreezeLoggerTests tests/test.cpp src/capture.cpp src/frames.cpp src/history.cpp src/storage.cpp)
target_link_libraries(FreezeLoggerTests PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32 tdh)
foreach(target FreezeLogger FreezeLoggerWatch FreezeLoggerTests)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
endforeach()
enable_testing()
add_executable(FreezeLoggerFeatures tests/features.cpp src/capture.cpp src/frames.cpp src/history.cpp src/storage.cpp src/lifetime.cpp src/context.cpp)
target_link_libraries(FreezeLoggerFeatures PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32 tdh d3d11 dxgi)
target_compile_options(FreezeLoggerFeatures PRIVATE /W4 /permissive- /EHsc)
target_compile_definitions(FreezeLoggerFeatures PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME history-storage COMMAND FreezeLoggerFeatures)
add_test(NAME real-dxgi-etw COMMAND FreezeLoggerFeatures --frames)
set_tests_properties(real-dxgi-etw PROPERTIES SKIP_RETURN_CODE 77)
add_executable(FreezeLoggerContextTests tests/context.cpp src/context.cpp src/capture.cpp src/history.cpp src/storage.cpp)
target_link_libraries(FreezeLoggerContextTests PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32)
target_compile_options(FreezeLoggerContextTests PRIVATE /W4 /permissive- /EHsc)
target_compile_definitions(FreezeLoggerContextTests PRIVATE _CRT_SECURE_NO_WARNINGS)
add_executable(FreezeLoggerLifetimeTests tests/lifetime.cpp src/lifetime.cpp)
target_link_libraries(FreezeLoggerLifetimeTests PRIVATE advapi32)
target_compile_options(FreezeLoggerLifetimeTests PRIVATE /W4 /permissive- /EHsc)
add_test(NAME engine-context COMMAND FreezeLoggerContextTests)
add_test(NAME collector-lifetime COMMAND FreezeLoggerLifetimeTests)
add_executable(FreezeLoggerSnapshotTests tests/snapshot_cleanup.cpp src/capture.cpp src/history.cpp src/storage.cpp)
target_link_libraries(FreezeLoggerSnapshotTests PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32)
target_compile_options(FreezeLoggerSnapshotTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeLoggerCaptureFallbackTests tests/capture_fallback.cpp src/capture.cpp src/history.cpp src/storage.cpp)
target_link_libraries(FreezeLoggerCaptureFallbackTests PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32)
target_compile_options(FreezeLoggerCaptureFallbackTests PRIVATE /W4 /permissive- /EHsc)
target_compile_definitions(FreezeLoggerCaptureFallbackTests PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME capture-fallback COMMAND FreezeLoggerCaptureFallbackTests)
set_tests_properties(capture-fallback PROPERTIES TIMEOUT 180)
add_executable(FreezeLoggerInitTests tests/initialization.cpp)
target_compile_options(FreezeLoggerInitTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeLoggerSuppressionTests tests/suppression.cpp)
target_compile_options(FreezeLoggerSuppressionTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeLoggerBuffoutTests tests/buffout.cpp)
target_compile_options(FreezeLoggerBuffoutTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeLoggerFuzzBuffout fuzz/fuzz_buffout_parser.cpp)
target_compile_options(FreezeLoggerFuzzBuffout PRIVATE /W4 /permissive- /EHsc)
add_test(NAME snapshot-cleanup COMMAND FreezeLoggerSnapshotTests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME release-verifier COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_verify_release.py)
  # Validates every shipped layout row against the executables present on this
  # machine; skips when none match, so it is harmless on a build agent.
  add_test(NAME layout-rows COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/verify_layout_rows.py)
  set_tests_properties(layout-rows PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
else()
  message(STATUS "Python3 not found: release-verifier and layout-rows tests are not registered")
endif()
add_executable(FreezeLoggerConsensusTests tests/consensus.cpp)
target_compile_options(FreezeLoggerConsensusTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeLoggerCrashTests tests/crash.cpp)
target_compile_options(FreezeLoggerCrashTests PRIVATE /W4 /permissive- /EHsc)
add_test(NAME crash-capture COMMAND FreezeLoggerCrashTests)
add_executable(FreezeLoggerExceptionTests tests/exceptions.cpp)
target_compile_options(FreezeLoggerExceptionTests PRIVATE /W4 /permissive- /EHsc)
add_test(NAME hang-suppression COMMAND FreezeLoggerSuppressionTests)
add_test(NAME thread-group-consensus COMMAND FreezeLoggerConsensusTests)
add_test(NAME exception-ring COMMAND FreezeLoggerExceptionTests)
add_test(NAME buffout-parser COMMAND FreezeLoggerBuffoutTests)
add_test(NAME buffout-fuzz-smoke COMMAND FreezeLoggerFuzzBuffout)
set_tests_properties(snapshot-cleanup PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME initialization-failures COMMAND FreezeLoggerInitTests $<TARGET_FILE:FreezeLogger> $<TARGET_FILE:FreezeLoggerWatch>)
add_test(NAME diagnostics COMMAND FreezeLoggerTests $<TARGET_FILE:FreezeLogger> $<TARGET_FILE:FreezeLoggerWatch>)
add_test(NAME oldest-f4se-fallback COMMAND FreezeLoggerTests $<TARGET_FILE:FreezeLogger> $<TARGET_FILE:FreezeLoggerWatch> --fallback)
set_tests_properties(history-storage real-dxgi-etw engine-context collector-lifetime snapshot-cleanup hang-suppression thread-group-consensus exception-ring crash-capture buffout-parser buffout-fuzz-smoke initialization-failures diagnostics oldest-f4se-fallback PROPERTIES TIMEOUT 180)
install(TARGETS FreezeLogger FreezeLoggerWatch RUNTIME DESTINATION F4SE/Plugins)
install(FILES dist/F4SE/Plugins/FreezeLogger.ini DESTINATION F4SE/Plugins)
if(FREELOGGER_SYMBOLS_MODE STREQUAL "required")
  install(FILES $<TARGET_PDB_FILE:FreezeLogger> $<TARGET_PDB_FILE:FreezeLoggerWatch> DESTINATION F4SE/Plugins)
endif()
install(FILES README.md LICENSE DESTINATION FreezeLogger)
install(FILES docs/ORIGIN.md docs/VALIDATION.md docs/CONTEXT-163.md docs/GAME-TESTS.md DESTINATION FreezeLogger/docs)
install(FILES src/telemetry.hpp DESTINATION FreezeLogger/include)
install(FILES build/sdk-evidence/abi-results.json DESTINATION FreezeLogger/docs OPTIONAL)
install(FILES build/context-image-evidence.json DESTINATION FreezeLogger/docs OPTIONAL)
install(FILES "build/validation-${PROJECT_VERSION}.xml" DESTINATION FreezeLogger/docs OPTIONAL)
install(FILES "build/changed-dll-${PROJECT_VERSION}.xml" DESTINATION FreezeLogger/docs OPTIONAL)
