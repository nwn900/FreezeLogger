cmake_minimum_required(VERSION 3.24)
project(FreezeSentryF4 VERSION 0.3.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0603)
add_library(FreezeSentryF4 SHARED src/plugin.cpp)
target_link_libraries(FreezeSentryF4 PRIVATE shell32 ole32)
add_executable(FreezeSentryF4Watch WIN32 src/watch.cpp src/capture.cpp src/frames.cpp src/history.cpp src/storage.cpp src/lifetime.cpp src/context.cpp)
target_link_libraries(FreezeSentryF4Watch PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32 tdh)
add_executable(FreezeSentryF4Tests tests/test.cpp src/capture.cpp src/frames.cpp src/history.cpp src/storage.cpp)
target_link_libraries(FreezeSentryF4Tests PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32 tdh)
foreach(target FreezeSentryF4 FreezeSentryF4Watch FreezeSentryF4Tests)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
endforeach()
enable_testing()
add_executable(FreezeSentryF4Features tests/features.cpp src/capture.cpp src/frames.cpp src/history.cpp src/storage.cpp src/lifetime.cpp src/context.cpp)
target_link_libraries(FreezeSentryF4Features PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32 tdh d3d11 dxgi)
target_compile_options(FreezeSentryF4Features PRIVATE /W4 /permissive- /EHsc)
target_compile_definitions(FreezeSentryF4Features PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME history-storage COMMAND FreezeSentryF4Features)
add_test(NAME real-dxgi-etw COMMAND FreezeSentryF4Features --frames)
set_tests_properties(real-dxgi-etw PROPERTIES SKIP_RETURN_CODE 77)
add_executable(FreezeSentryF4ContextTests tests/context.cpp src/context.cpp src/capture.cpp src/history.cpp src/storage.cpp)
target_link_libraries(FreezeSentryF4ContextTests PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32)
target_compile_options(FreezeSentryF4ContextTests PRIVATE /W4 /permissive- /EHsc)
target_compile_definitions(FreezeSentryF4ContextTests PRIVATE _CRT_SECURE_NO_WARNINGS)
add_executable(FreezeSentryF4LifetimeTests tests/lifetime.cpp src/lifetime.cpp)
target_link_libraries(FreezeSentryF4LifetimeTests PRIVATE advapi32)
target_compile_options(FreezeSentryF4LifetimeTests PRIVATE /W4 /permissive- /EHsc)
add_test(NAME engine-context COMMAND FreezeSentryF4ContextTests)
add_test(NAME collector-lifetime COMMAND FreezeSentryF4LifetimeTests)
add_executable(FreezeSentryF4SnapshotTests tests/snapshot_cleanup.cpp src/capture.cpp src/history.cpp src/storage.cpp)
target_link_libraries(FreezeSentryF4SnapshotTests PRIVATE dbghelp psapi shell32 ole32 bcrypt version advapi32)
target_compile_options(FreezeSentryF4SnapshotTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeSentryF4InitTests tests/initialization.cpp)
target_compile_options(FreezeSentryF4InitTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeSentryF4SuppressionTests tests/suppression.cpp)
target_compile_options(FreezeSentryF4SuppressionTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeSentryF4BuffoutTests tests/buffout.cpp)
target_compile_options(FreezeSentryF4BuffoutTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeSentryF4FuzzBuffout fuzz/fuzz_buffout_parser.cpp)
target_compile_options(FreezeSentryF4FuzzBuffout PRIVATE /W4 /permissive- /EHsc)
add_test(NAME snapshot-cleanup COMMAND FreezeSentryF4SnapshotTests)
add_executable(FreezeSentryF4ConsensusTests tests/consensus.cpp)
target_compile_options(FreezeSentryF4ConsensusTests PRIVATE /W4 /permissive- /EHsc)
add_executable(FreezeSentryF4ExceptionTests tests/exceptions.cpp)
target_compile_options(FreezeSentryF4ExceptionTests PRIVATE /W4 /permissive- /EHsc)
add_test(NAME hang-suppression COMMAND FreezeSentryF4SuppressionTests)
add_test(NAME thread-group-consensus COMMAND FreezeSentryF4ConsensusTests)
add_test(NAME exception-ring COMMAND FreezeSentryF4ExceptionTests)
add_test(NAME buffout-parser COMMAND FreezeSentryF4BuffoutTests)
add_test(NAME buffout-fuzz-smoke COMMAND FreezeSentryF4FuzzBuffout)
set_tests_properties(snapshot-cleanup PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME initialization-failures COMMAND FreezeSentryF4InitTests $<TARGET_FILE:FreezeSentryF4> $<TARGET_FILE:FreezeSentryF4Watch>)
add_test(NAME diagnostics COMMAND FreezeSentryF4Tests $<TARGET_FILE:FreezeSentryF4> $<TARGET_FILE:FreezeSentryF4Watch>)
add_test(NAME oldest-f4se-fallback COMMAND FreezeSentryF4Tests $<TARGET_FILE:FreezeSentryF4> $<TARGET_FILE:FreezeSentryF4Watch> --fallback)
set_tests_properties(history-storage real-dxgi-etw engine-context collector-lifetime snapshot-cleanup hang-suppression thread-group-consensus exception-ring buffout-parser buffout-fuzz-smoke initialization-failures diagnostics oldest-f4se-fallback PROPERTIES TIMEOUT 180)
install(TARGETS FreezeSentryF4 FreezeSentryF4Watch RUNTIME DESTINATION F4SE/Plugins)
install(FILES dist/F4SE/Plugins/FreezeSentryF4.ini DESTINATION F4SE/Plugins)
install(FILES $<TARGET_PDB_FILE:FreezeSentryF4> $<TARGET_PDB_FILE:FreezeSentryF4Watch> DESTINATION F4SE/Plugins OPTIONAL)
install(FILES README.md LICENSE DESTINATION FreezeSentryF4)
install(FILES docs/ORIGIN.md docs/VALIDATION.md docs/CONTEXT-163.md docs/GAME-TESTS.md DESTINATION FreezeSentryF4/docs)
install(FILES src/telemetry.hpp DESTINATION FreezeSentryF4/include)
install(FILES build/sdk-evidence/abi-results.json DESTINATION FreezeSentryF4/docs OPTIONAL)
install(FILES build/context-image-evidence.json DESTINATION FreezeSentryF4/docs OPTIONAL)
install(FILES "build/validation-${PROJECT_VERSION}.xml" DESTINATION FreezeSentryF4/docs OPTIONAL)
install(FILES "build/changed-dll-${PROJECT_VERSION}.xml" DESTINATION FreezeSentryF4/docs OPTIONAL)
